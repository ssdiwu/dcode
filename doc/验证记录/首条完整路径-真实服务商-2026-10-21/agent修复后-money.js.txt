/**
 * 金额取整到分（四舍五入）。
 *
 * 约定：金额一律以「元」为单位的十进制数表示，所有金额边界都必须经过本函数，
 * 调用方不得自行取整。
 *
 * 实现说明：
 * 1. 先把「元」放大 100 倍换成「分」，按 Math.round 取整，再换回「元」；
 * 2. 放大后先用 toFixed(8) 抹平二进制浮点的表示误差，否则 1.005 * 100
 *    会得到 100.49999999999999，被少舍入一分。
 */
export function roundToCents(value) {
  const scaled = Number((value * 100).toFixed(8));
  // 末尾加 0，把 -0 归一成 0，避免下游比较时 -0 与 0 不相等。
  return Math.round(scaled) / 100 + 0;
}
